# ============================================================
# ETEL Delivery & Ride - browser security headers (Week 1 hardening)
#
# HOW TO USE (cPanel File Manager, no SSH):
#   1. Open public_html/delivery/.htaccess  (create it if it does not exist).
#      If it already has content, APPEND this block - do NOT replace the file.
#   2. Save. Open each app (customer, ride, driver, courier, service, admin)
#      with the browser DevTools Console open and use it normally.
#   3. Any line that says "Refused to ... because it violates the following
#      Content Security Policy" is something this policy WOULD block. It is
#      only reported (not blocked) while the header is "Report-Only".
#   4. When a full day of normal use shows no violations, change the header
#      name below from Content-Security-Policy-Report-Only to
#      Content-Security-Policy to start enforcing.
#
# What the policy does:
#   - Blocks scripts/styles/fonts from sites that are not on the list below.
#   - connect-src limits where the page can SEND data, so a stolen login token
#     cannot be posted to an attacker's server even if an XSS bug slips in.
#   - Stops other websites from framing the apps (clickjacking).
#   'unsafe-inline' stays for scripts because the apps use inline handlers
#   (onclick=...) heavily; removing it needs a refactor, see SECURITY_WEEK1.md.
# ============================================================
<IfModule mod_headers.c>
  Header always set Content-Security-Policy-Report-Only "default-src 'self'; script-src 'self' 'unsafe-inline' https://maps.googleapis.com https://maps.gstatic.com https://cdnjs.cloudflare.com https://cdn.jsdelivr.net https://www.google.com; style-src 'self' 'unsafe-inline' https://fonts.googleapis.com https://cdnjs.cloudflare.com https://cdn.jsdelivr.net; font-src 'self' data: https://fonts.gstatic.com https://cdnjs.cloudflare.com https://cdn.jsdelivr.net; img-src 'self' data: blob: https:; connect-src 'self' https://ethelmarket.com https://www.ethelmarket.com https://maps.googleapis.com https://*.googleapis.com https://nominatim.openstreetmap.org; frame-src 'self' https://www.google.com https://maps.google.com; worker-src 'self' blob:; manifest-src 'self'; object-src 'none'; base-uri 'self'; form-action 'self'; frame-ancestors 'self' https://ethelmarket.com https://www.ethelmarket.com"

  # Only add HSTS once you are sure EVERY page on this domain works over https.
  Header always set Strict-Transport-Security "max-age=31536000"

  Header always set X-Content-Type-Options "nosniff"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set Permissions-Policy "geolocation=(self), camera=(self), microphone=(), payment=(self), usb=()"
</IfModule>
